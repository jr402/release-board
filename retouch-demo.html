RetouchTool — Liquify + Transform

Ouvre une image, retouche-la, puis Enregistrer : le résultat s'affiche à droite. Le module est un seul fichier : retouch-tool.js.

ou Ctrl+V une image
Dépose une image ici

Intégration

<script src="retouch-tool.js"></script>
<script>
RetouchTool.open({
  src: imageUrlOuDataUrl,          // URL, data URL, <img>, <canvas>, File / Blob
  title: "Nom de l'image",
  format: "auto",                  // "image/png" | "image/jpeg" | "image/webp"
  lang: "fr",
  onSave(result) {                 // { dataUrl, blob, mime, width, height, canvas }
    monImage.src = result.dataUrl; // peut renvoyer une Promise (upload…) : l'outil attend
  },
});
</script>
